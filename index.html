// IBI Settlement Tracker — Google Apps Script Backend
// Paste this into script.google.com → New Project → Deploy as Web App

const SHEET_ID  = 'YOUR_GOOGLE_SHEET_ID_HERE';
const SHEET_TAB = 'Settlements';

const HEADERS = [
  'Timestamp','Platform','Order ID','Order Date','Product',
  'SKU','Qty','Selling Price','Fees','Net Settlement',
  'Settlement Date','Status','Days Since Order','Remarks'
];

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const ss    = SpreadsheetApp.openById(SHEET_ID);
    let   sheet = ss.getSheetByName(SHEET_TAB);

    if (!sheet) {
      sheet = ss.insertSheet(SHEET_TAB);
      sheet.getRange(1, 1, 1, HEADERS.length).setValues([HEADERS]);
      sheet.getRange(1, 1, 1, HEADERS.length)
           .setBackground('#0D1B2A').setFontColor('#00D4F0')
           .setFontWeight('bold');
      sheet.setFrozenRows(1);
    }

    const rows = payload.rows || [];
    const ts   = new Date().toISOString();

    const data = rows.map(r => [
      ts,
      r.platform        || '',
      r.orderId         || '',
      r.orderDate       || '',
      r.product         || '',
      r.sku             || '',
      r.qty             || '',
      r.sellingPrice    || 0,
      r.fees            || 0,
      r.netSettlement   || 0,
      r.settlementDate  || '',
      r.status          || '',
      r.daysSinceOrder  || '',
      r.remarks         || ''
    ]);

    if (data.length > 0) {
      const lastRow = sheet.getLastRow();
      sheet.getRange(lastRow + 1, 1, data.length, HEADERS.length)
           .setValues(data);

      // Colour-code status column (col 12)
      data.forEach((row, i) => {
        const cell = sheet.getRange(lastRow + 1 + i, 12);
        const status = row[11];
        if (status === 'Settled')   cell.setBackground('#d4edda').setFontColor('#155724');
        if (status === 'Pending')   cell.setBackground('#fff3cd').setFontColor('#856404');
        if (status === 'Returned')  cell.setBackground('#f8d7da').setFontColor('#721c24');
        if (status === 'Cancelled') cell.setBackground('#e2e3e5').setFontColor('#383d41');
      });
    }

    return ContentService
      .createTextOutput(JSON.stringify({ ok: true, rows: data.length }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({ ok: false, error: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService
    .createTextOutput(JSON.stringify({ ok: true, msg: 'IBI Settlement Tracker GAS is live' }))
    .setMimeType(ContentService.MimeType.JSON);
}
